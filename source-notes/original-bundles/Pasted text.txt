mport React, { useEffect, useState, useCallback } from 'react';
import { Radar, Loader2, ChevronDown, CheckCircle2, AlertCircle, Lightbulb } from 'lucide-react';
import { base44 } from '@/api/base44Client';

export default function RadarPage() {
  const [lectures, setLectures] = useState([]);
  const [loading, setLoading] = useState(true);
  const [selectedId, setSelectedId] = useState('');
  const [scanning, setScanning] = useState(false);
  const [concepts, setConcepts] = useState(null);
  const [error, setError] = useState('');

  const loadLectures = useCallback(async () => {
    try {
      const { items } = await base44.entities.Lecture.filter({}, { sort: '-created_date', limit: 100 });
      setLectures(items);
    } catch (err) {
      console.error(err);
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    loadLectures();
  }, [loadLectures]);

  const selectedLecture = lectures.find((l) => l.id === selectedId);

  const handleScan = async () => {
    if (!selectedLecture) return;
    setScanning(true);
    setError('');
    setConcepts(null);
    try {
      const res = await base44.functions.invoke('lecture_ai', {
        action: 'confusion_scan',
        lectureId: selectedLecture.id,
        content: selectedLecture.content,
        title: selectedLecture.title,
      });
      const result = res.data?.result;
      if (result?.concepts) {
        setConcepts(result.concepts);
      } else if (typeof result === 'string') {
        try {
          const parsed = JSON.parse(result);
          setConcepts(parsed.concepts || []);
        } catch {
          setError('Could not parse scan results.');
        }
      } else {
        setError('No concepts returned.');
      }
    } catch (err) {
      setError(err.message || 'Scan failed.');
    } finally {
      setScanning(false);
    }
  };

  return (
    <div className="p-8 max-w-4xl mx-auto">
      <div className="mb-6">
        <div className="flex items-center gap-2 mb-1">
          <Radar className="w-6 h-6 text-indigo-600" />
          <h1 className="text-2xl font-bold text-slate-900">Confusion Radar</h1>
        </div>
        <p className="text-slate-500 text-sm">
          AI scans a lecture transcript for concepts students typically find confusing, with explanations and practice questions.
        </p>
      </div>

      <div className="bg-white rounded-2xl border border-slate-200 p-5 mb-6">
        <label className="text-sm font-medium text-slate-700 mb-2 block">Select a lecture to scan</label>
        <div className="flex gap-3">
          <div className="relative flex-1">
            <select
              value={selectedId}
              onChange={(e) => { setSelectedId(e.target.value); setConcepts(null); setError(''); }}
              className="w-full appearance-none bg-slate-50 border border-slate-200 rounded-xl px-4 py-2.5 pr-10 text-sm text-slate-900 focus:outline-none focus:border-indigo-400 focus:ring-2 focus:ring-indigo-100 transition-all"
            >
              <option value="">Choose a lecture…</option>
              {lectures.map((l) => (
                <option key={l.id} value={l.id}>{l.title}</option>
              ))}
            </select>
            <ChevronDown className="w-4 h-4 text-slate-400 absolute right-3.5 top-1/2 -translate-y-1/2 pointer-events-none" />
          </div>
          <button
            onClick={handleScan}
            disabled={!selectedId || scanning}
            className="inline-flex items-center gap-2 bg-indigo-600 hover:bg-indigo-700 disabled:opacity-50 text-white px-5 py-2.5 rounded-xl text-sm font-medium transition-colors whitespace-nowrap"
          >
            {scanning ? <Loader2 className="w-4 h-4 animate-spin" /> : <Radar className="w-4 h-4" />}
            {scanning ? 'Scanning…' : 'Scan for confusion'}
          </button>
        </div>
        {selectedLecture && !selectedLecture.content && (
          <p className="text-xs text-amber-600 mt-2">This lecture has no transcript content to scan.</p>
        )}
      </div>

      {error && (
        <div className="bg-red-50 border border-red-200 rounded-xl p-4 text-sm text-red-700 mb-4 flex items-center gap-2">
          <AlertCircle className="w-4 h-4 shrink-0" />
          {error}
        </div>
      )}

      {scanning && (
        <div className="bg-white rounded-2xl border border-slate-200 p-12 text-center">
          <Loader2 className="w-8 h-8 text-indigo-500 animate-spin mx-auto mb-3" />
          <p className="text-sm text-slate-500">Analyzing transcript for confusing concepts…</p>
        </div>
      )}

      {concepts && concepts.length > 0 && (
        <div className="space-y-4">
          <div className="flex items-center gap-2 text-sm font-medium text-slate-700">
            <CheckCircle2 className="w-4 h-4 text-emerald-500" />
            Found {concepts.length} potentially confusing concept{concepts.length !== 1 ? 's' : ''}
          </div>
          {concepts.map((concept, i) => (
            <ConceptCard key={i} concept={concept} index={i} />
          ))}
        </div>
      )}

      {concepts && concepts.length === 0 && !error && (
        <div className="bg-white rounded-2xl border border-slate-200 p-8 text-center text-sm text-slate-500">
          No confusing concepts detected.
        </div>
      )}

      {!loading && lectures.length === 0 && (
        <div className="bg-white rounded-2xl border border-dashed border-slate-300 p-12 text-center">
          <p className="text-slate-500 text-sm">Add a lecture first to use the Confusion Radar.</p>
        </div>
      )}
    </div>
  );
}

function ConceptCard({ concept, index }) {
  const [showAnswer, setShowAnswer] = useState(false);
  return (
    <div className="bg-white rounded-2xl border border-slate-200 overflow-hidden">
      <div className="p-5">
        <div className="flex items-start gap-3 mb-3">
          <div className="w-7 h-7 rounded-full bg-amber-100 text-amber-700 flex items-center justify-center text-xs font-bold shrink-0">
            {index + 1}
          </div>
          <div className="flex-1">
            <h3 className="font-semibold text-slate-900 text-base">{concept.concept}</h3>
            <p className="text-sm text-slate-500 mt-1">{concept.why_confusing}</p>
          </div>
        </div>
        <div className="bg-indigo-50 rounded-xl p-4 flex gap-2.5">
          <Lightbulb className="w-4 h-4 text-indigo-500 shrink-0 mt-0.5" />
          <p className="text-sm text-slate-700 leading-relaxed">{concept.explanation}</p>
        </div>
        {concept.practice_question && (
          <div className="mt-4 border-t border-slate-100 pt-4">
            <p className="text-sm font-medium text-slate-900 mb-3">{concept.practice_question}</p>
            <div className="space-y-2">
              {concept.options?.map((opt, i) => {
                const isCorrect = String.fromCharCode(65 + i) === concept.correct_answer;
                return (
                  <button
                    key={i}
                    onClick={() => setShowAnswer(true)}
                    className={`w-full text-left flex items-center gap-2.5 px-3.5 py-2.5 rounded-lg text-sm border transition-all ${
                      showAnswer && isCorrect
                        ? 'bg-emerald-50 border-emerald-300 text-emerald-800'
                        : showAnswer
                        ? 'bg-slate-50 border-slate-200 text-slate-600'
                        : 'bg-slate-50 border-slate-200 text-slate-700 hover:border-indigo-300'
                    }`}
                  >
                    <span className="font-semibold text-xs w-5">{String.fromCharCode(65 + i)}.</span>
                    {opt}
                    {showAnswer && isCorrect && <CheckCircle2 className="w-4 h-4 text-emerald-500 ml-auto" />}
                  </button>
                );
              })}
            </div>
            {showAnswer && (
              <p className="text-xs text-emerald-600 mt-2 font-medium">
                Correct answer: {concept.correct_answer}
              </p>
            )}
            {!showAnswer && (
              <p className="text-xs text-slate-400 mt-2">Click an option to reveal the answer.</p>
            )}
          </div>
        )}
      </div>
    </div>
  );
}     this is radar page
