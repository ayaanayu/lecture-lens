// PlannerPage.jsx — Mission Planner with calendar, tasks, and milestone progress
import React, { useState, useEffect } from 'react';
import { Calendar, Plus, Trash2, Target, Loader2, CheckCircle2 } from 'lucide-react';
import { DragDropContext, Droppable, Draggable } from '@hello-pangea/dnd';
import { base44 } from '@/api/base44Client';
import {
  format, startOfMonth, endOfMonth, startOfWeek, endOfWeek,
  addDays, isSameMonth, isSameDay, addMonths, subMonths, parseISO
} from 'date-fns';

// Priority badge colors
const priorityColors = {
  high: 'bg-red-100 text-red-700',
  medium: 'bg-amber-100 text-amber-700',
  low: 'bg-emerald-100 text-emerald-700',
};

export default function PlannerPage() {
  const [tasks, setTasks] = useState([]);
  const [loading, setLoading] = useState(true);
  const [currentMonth, setCurrentMonth] = useState(new Date());
  const [showForm, setShowForm] = useState(false);
  const [newTask, setNewTask] = useState({
    title: '', subject: '', priority: 'medium', due_date: '', milestone: ''
  });

  useEffect(() => {
    loadTasks();
  }, []);

  // Load all tasks sorted by due date
  const loadTasks = async () => {
    try {
      const { items } = await base44.entities.Task.filter({}, { sort: 'due_date', limit: 100 });
      setTasks(items);
    } catch (err) {
      console.error(err);
    } finally {
      setLoading(false);
    }
  };

  // Create a new task
  const addTask = async () => {
    if (!newTask.title.trim()) return;
    try {
      const created = await base44.entities.Task.create({
        ...newTask,
        title: newTask.title.trim(),
        status: 'todo',
        progress: 0,
      });
      setTasks([...tasks, created]);
      setNewTask({ title: '', subject: '', priority: 'medium', due_date: '', milestone: '' });
      setShowForm(false);
    } catch (err) {
      console.error(err);
    }
  };

  // Cycle task status: todo → in_progress → done
  const toggleStatus = async (task) => {
    const next = task.status === 'todo' ? 'in_progress' : task.status === 'in_progress' ? 'done' : 'todo';
    const progress = next === 'done' ? 100 : next === 'in_progress' ? 50 : 0;
    try {
      await base44.entities.Task.update(task.id, { status: next, progress });
      setTasks(tasks.map((t) => (t.id === task.id ? { ...t, status: next, progress } : t)));
    } catch (err) {
      console.error(err);
    }
  };

  // Delete a task
  const deleteTask = async (id) => {
    try {
      await base44.entities.Task.delete(id);
      setTasks(tasks.filter((t) => t.id !== id));
    } catch (err) {
      console.error(err);
    }
  };

  // Handle drag-and-drop between status columns
  const onDragEnd = async (result) => {
    const { source, destination } = result;
    if (!destination || source.droppableId === destination.droppableId) return;
    const draggedTask = tasks.find((t) => t.id === result.draggableId);
    if (!draggedTask) return;
    const newStatus = destination.droppableId;
    const progress = newStatus === 'done' ? 100 : newStatus === 'in_progress' ? 50 : 0;
    try {
      await base44.entities.Task.update(draggedTask.id, { status: newStatus, progress });
      setTasks(tasks.map((t) => (t.id === draggedTask.id ? { ...t, status: newStatus, progress } : t)));
    } catch (err) {
      console.error(err);
    }
  };

  // Build calendar grid for current month
  const monthStart = startOfMonth(currentMonth);
  const monthEnd = endOfMonth(currentMonth);
  const calStart = startOfWeek(monthStart);
  const calEnd = endOfWeek(monthEnd);
  const calendarDays = [];
  let day = calStart;
  while (day <= calEnd) {
    calendarDays.push(day);
    day = addDays(day, 1);
  }

  // Group tasks by due date for calendar dots
  const tasksByDate = {};
  tasks.forEach((t) => {
    if (t.due_date) {
      const key = format(parseISO(t.due_date), 'yyyy-MM-dd');
      if (!tasksByDate[key]) tasksByDate[key] = [];
      tasksByDate[key].push(t);
    }
  });

  // Compute milestone progress from tasks
  const milestones = {};
  tasks.forEach((t) => {
    if (t.milestone) {
      if (!milestones[t.milestone]) milestones[t.milestone] = { total: 0, done: 0, progress: 0 };
      milestones[t.milestone].total++;
      if (t.status === 'done') milestones[t.milestone].done++;
    }
  });
  Object.values(milestones).forEach((m) => {
    m.progress = m.total > 0 ? Math.round((m.done / m.total) * 100) : 0;
  });

  return (
    <div className="p-8 max-w-5xl mx-auto">
      {/* Header */}
      <div className="flex items-center justify-between mb-6 flex-wrap gap-3">
        <div>
          <div className="flex items-center gap-2 mb-1">
            <Target className="w-5 h-5 text-indigo-600" />
            <h1 className="text-2xl font-bold text-slate-900">Mission Planner</h1>
          </div>
          <p className="text-sm text-slate-500">Track tasks, deadlines, and study milestones.</p>
        </div>
        <button
          onClick={() => setShowForm(!showForm)}
          className="inline-flex items-center gap-2 bg-indigo-600 hover:bg-indigo-700 text-white px-4 py-2 rounded-lg text-sm font-medium transition-colors"
        >
          <Plus className="w-4 h-4" />
          Add Task
        </button>
      </div>

      {/* Add task form */}
      {showForm && (
        <div className="bg-white rounded-2xl border border-slate-200 p-5 mb-6 space-y-3">
          <input
            value={newTask.title}
            onChange={(e) => setNewTask({ ...newTask, title: e.target.value })}
            placeholder="Task title…"
            className="w-full text-sm px-3 py-2 bg-slate-50 rounded-lg border border-slate-200 focus:outline-none focus:border-indigo-400"
          />
          <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
            <input
              value={newTask.subject}
              onChange={(e) => setNewTask({ ...newTask, subject: e.target.value })}
              placeholder="Subject"
              className="text-sm px-3 py-2 bg-slate-50 rounded-lg border border-slate-200 focus:outline-none focus:border-indigo-400"
            />
            <select
              value={newTask.priority}
              onChange={(e) => setNewTask({ ...newTask, priority: e.target.value })}
              className="text-sm px-3 py-2 bg-slate-50 rounded-lg border border-slate-200 focus:outline-none focus:border-indigo-400"
            >
              <option value="low">Low</option>
              <option value="medium">Medium</option>
              <option value="high">High</option>
            </select>
            <input
              type="date"
              value={newTask.due_date}
              onChange={(e) => setNewTask({ ...newTask, due_date: e.target.value })}
              className="text-sm px-3 py-2 bg-slate-50 rounded-lg border border-slate-200 focus:outline-none focus:border-indigo-400"
            />
            <input
              value={newTask.milestone}
              onChange={(e) => setNewTask({ ...newTask, milestone: e.target.value })}
              placeholder="Milestone"
              className="text-sm px-3 py-2 bg-slate-50 rounded-lg border border-slate-200 focus:outline-none focus:border-indigo-400"
            />
          </div>
          <button
            onClick={addTask}
            className="bg-indigo-600 hover:bg-indigo-700 text-white px-4 py-2 rounded-lg text-sm font-medium transition-colors"
          >
            Save Task
          </button>
        </div>
      )}

      <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
        {/* Calendar */}
        <div className="bg-white rounded-2xl border border-slate-200 p-5">
          <div className="flex items-center justify-between mb-4">
            <h3 className="text-sm font-semibold text-slate-700 flex items-center gap-2">
              <Calendar className="w-4 h-4 text-indigo-600" />
              {format(currentMonth, 'MMMM yyyy')}
            </h3>
            <div className="flex gap-1">
              <button onClick={() => setCurrentMonth(subMonths(currentMonth, 1))} className="px-2 py-1 rounded-lg hover:bg-slate-100 text-slate-500 text-sm">←</button>
              <button onClick={() => setCurrentMonth(addMonths(currentMonth, 1))} className="px-2 py-1 rounded-lg hover:bg-slate-100 text-slate-500 text-sm">→</button>
            </div>
          </div>
          {/* Day labels */}
          <div className="grid grid-cols-7 gap-1 mb-1">
            {['S', 'M', 'T', 'W', 'T', 'F', 'S'].map((d, i) => (
              <div key={i} className="text-center text-xs font-semibold text-slate-400 py-1">{d}</div>
            ))}
          </div>
          {/* Calendar grid */}
          <div className="grid grid-cols-7 gap-1">
            {calendarDays.map((d) => {
              const key = format(d, 'yyyy-MM-dd');
              const dayTasks = tasksByDate[key] || [];
              const inMonth = isSameMonth(d, currentMonth);
              const isToday = isSameDay(d, new Date());
              return (
                <div
                  key={key}
                  className={`min-h-[44px] p-1 rounded-lg text-center text-xs ${
                    inMonth ? 'bg-slate-50' : 'bg-transparent text-slate-300'
                  } ${isToday ? 'ring-2 ring-indigo-400' : ''}`}
                >
                  <div className={`font-medium ${isToday ? 'text-indigo-600' : 'text-slate-600'}`}>
                    {format(d, 'd')}
                  </div>
                  {dayTasks.length > 0 && (
                    <div className="flex justify-center gap-0.5 mt-0.5">
                      {dayTasks.slice(0, 3).map((t, i) => (
                        <div
                          key={i}
                          className={`w-1.5 h-1.5 rounded-full ${
                            t.priority === 'high' ? 'bg-red-400' :
                            t.priority === 'medium' ? 'bg-amber-400' : 'bg-emerald-400'
                          }`}
                        />
                      ))}
                    </div>
                  )}
                </div>
              );
            })}
          </div>
        </div>

        {/* Milestone progress */}
        <div className="bg-white rounded-2xl border border-slate-200 p-5">
          <h3 className="text-sm font-semibold text-slate-700 mb-4 flex items-center gap-2">
            <Target className="w-4 h-4 text-indigo-600" />
            Milestone Progress
          </h3>
          {Object.keys(milestones).length === 0 ? (
            <p className="text-sm text-slate-400 text-center py-6">
              No milestones yet. Add a milestone name when creating tasks.
            </p>
          ) : (
            <div className="space-y-3">
              {Object.entries(milestones).map(([name, m]) => (
                <div key={name}>
                  <div className="flex items-center justify-between mb-1">
                    <span className="text-sm font-medium text-slate-700">{name}</span>
                    <span className="text-xs text-slate-500">{m.done}/{m.total}</span>
                  </div>
                  <div className="w-full bg-slate-100 rounded-full h-2">
                    <div
                      className="bg-indigo-600 h-2 rounded-full transition-all duration-500"
                      style={{ width: `${m.progress}%` }}
                    />
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>
      </div>

      {/* Categorized drag-and-drop task lists */}
      <div className="mt-6">
        <h3 className="text-sm font-semibold text-slate-700 mb-3">Tasks — drag between columns</h3>
        {loading ? (
          <div className="text-center py-6">
            <Loader2 className="w-6 h-6 text-indigo-500 animate-spin mx-auto" />
          </div>
        ) : tasks.length === 0 ? (
          <div className="bg-white rounded-2xl border border-dashed border-slate-300 p-8 text-center">
            <Target className="w-10 h-10 text-slate-300 mx-auto mb-2" />
            <p className="text-sm text-slate-400">No tasks yet. Click "Add Task" to start planning.</p>
          </div>
        ) : (
          <DragDropContext onDragEnd={onDragEnd}>
            <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
              {/* To Do column */}
              <Droppable droppableId="todo">
                {(provided, snapshot) => (
                  <div
                    ref={provided.innerRef}
                    {...provided.droppableProps}
                    className={`rounded-2xl border p-4 min-h-[200px] ${snapshot.isDraggingOver ? 'border-indigo-400 bg-indigo-50' : 'border-slate-200 bg-slate-50'}`}
                  >
                    <h4 className="text-xs font-semibold text-slate-500 uppercase tracking-wider mb-3">To Do</h4>
                    <div className="space-y-2">
                      {tasks.filter((t) => t.status === 'todo').map((task, index) => (
                        <Draggable key={task.id} draggableId={task.id} index={index}>
                          {(prov, snap) => (
                            <div
                              ref={prov.innerRef}
                              {...prov.draggableProps}
                              {...prov.dragHandleProps}
                              className={`bg-white rounded-xl border p-3 ${snap.isDragging ? 'border-indigo-400 shadow-lg' : 'border-slate-200'}`}
                            >
                              <div className="flex items-start justify-between gap-2">
                                <p className="text-sm font-medium text-slate-800">{task.title}</p>
                                <button onClick={() => deleteTask(task.id)} className="text-slate-300 hover:text-red-500 transition-colors shrink-0">
                                  <Trash2 className="w-3.5 h-3.5" />
                                </button>
                              </div>
                              <div className="flex items-center gap-2 mt-1.5">
                                {task.subject && <span className="text-xs text-slate-400">{task.subject}</span>}
                                {task.due_date && <span className="text-xs text-slate-400">• {format(parseISO(task.due_date), 'MMM d')}</span>}
                                <span className={`text-xs font-medium px-2 py-0.5 rounded-full ml-auto ${priorityColors[task.priority] || priorityColors.medium}`}>
                                  {task.priority}
                                </span>
                              </div>
                            </div>
                          )}
                        </Draggable>
                      ))}
                      {provided.placeholder}
                    </div>
                  </div>
                )}
              </Droppable>

              {/* In Progress column */}
              <Droppable droppableId="in_progress">
                {(provided, snapshot) => (
                  <div
                    ref={provided.innerRef}
                    {...provided.droppableProps}
                    className={`rounded-2xl border p-4 min-h-[200px] ${snapshot.isDraggingOver ? 'border-indigo-400 bg-indigo-50' : 'border-slate-200 bg-slate-50'}`}
                  >
                    <h4 className="text-xs font-semibold text-indigo-500 uppercase tracking-wider mb-3">In Progress</h4>
                    <div className="space-y-2">
                      {tasks.filter((t) => t.status === 'in_progress').map((task, index) => (
                        <Draggable key={task.id} draggableId={task.id} index={index}>
                          {(prov, snap) => (
                            <div
                              ref={prov.innerRef}
                              {...prov.draggableProps}
                              {...prov.dragHandleProps}
                              className={`bg-white rounded-xl border p-3 ${snap.isDragging ? 'border-indigo-400 shadow-lg' : 'border-slate-200'}`}
                            >
                              <div className="flex items-start justify-between gap-2">
                                <p className="text-sm font-medium text-slate-800">{task.title}</p>
                                <button onClick={() => deleteTask(task.id)} className="text-slate-300 hover:text-red-500 transition-colors shrink-0">
                                  <Trash2 className="w-3.5 h-3.5" />
                                </button>
                              </div>
                              <div className="flex items-center gap-2 mt-1.5">
                                {task.subject && <span className="text-xs text-slate-400">{task.subject}</span>}
                                {task.due_date && <span className="text-xs text-slate-400">• {format(parseISO(task.due_date), 'MMM d')}</span>}
                                <span className={`text-xs font-medium px-2 py-0.5 rounded-full ml-auto ${priorityColors[task.priority] || priorityColors.medium}`}>
                                  {task.priority}
                                </span>
                              </div>
                            </div>
                          )}
                        </Draggable>
                      ))}
                      {provided.placeholder}
                    </div>
                  </div>
                )}
              </Droppable>

              {/* Done column */}
              <Droppable droppableId="done">
                {(provided, snapshot) => (
                  <div
                    ref={provided.innerRef}
                    {...provided.droppableProps}
                    className={`rounded-2xl border p-4 min-h-[200px] ${snapshot.isDraggingOver ? 'border-indigo-400 bg-indigo-50' : 'border-slate-200 bg-slate-50'}`}
                  >
                    <h4 className="text-xs font-semibold text-emerald-500 uppercase tracking-wider mb-3">Done</h4>
                    <div className="space-y-2">
                      {tasks.filter((t) => t.status === 'done').map((task, index) => (
                        <Draggable key={task.id} draggableId={task.id} index={index}>
                          {(prov, snap) => (
                            <div
                              ref={prov.innerRef}
                              {...prov.draggableProps}
                              {...prov.dragHandleProps}
                              className={`bg-white rounded-xl border p-3 ${snap.isDragging ? 'border-indigo-400 shadow-lg' : 'border-slate-200'}`}
                            >
                              <div className="flex items-start justify-between gap-2">
                                <p className="text-sm font-medium text-slate-400 line-through">{task.title}</p>
                                <button onClick={() => deleteTask(task.id)} className="text-slate-300 hover:text-red-500 transition-colors shrink-0">
                                  <Trash2 className="w-3.5 h-3.5" />
                                </button>
                              </div>
                              <div className="flex items-center gap-2 mt-1.5">
                                {task.subject && <span className="text-xs text-slate-400">{task.subject}</span>}
                                <span className={`text-xs font-medium px-2 py-0.5 rounded-full ml-auto ${priorityColors[task.priority] || priorityColors.medium}`}>
                                  {task.priority}
                                </span>
                              </div>
                            </div>
                          )}
                        </Draggable>
                      ))}
                      {provided.placeholder}
                    </div>
                  </div>
                )}
              </Droppable>
            </div>
          </DragDropContext>
        )}
      </div>
    </div>
  );
}     this is planner page
